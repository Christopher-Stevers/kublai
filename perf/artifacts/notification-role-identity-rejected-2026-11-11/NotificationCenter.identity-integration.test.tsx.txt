const state = vi.hoisted(() => ({
  authority: {} as object,
  identity: vi.fn<() => Promise<{ accountKey: string | null }>>(),
  inputs: [] as string[],
}));
vi.mock("~/components/jobs/DrawingAccountProvider", () => ({

vi.mock("~/lib/employee-simulation", () => ({
  simulationViewHeaders: () => ({ view: "user" }),
}));
vi.mock("~/hooks/use-online-status", () => ({ useOnlineStatus: () => true }));
vi.mock("~/lib/app-badge", () => ({ updateAppBadge: vi.fn() }));
vi.mock("./use-notification-reminder", () => ({
  useNotificationReminder: () => 0,
}));
vi.mock("next/navigation", () => ({ useRouter: () => ({ push: vi.fn() }) }));
vi.mock("@trpc/react-query", () => ({
  getQueryKey: (proc: { key: string }, input: unknown) => [
    proc.key,
    input ?? null,
  ],
}));
vi.mock("~/trpc/react", () => ({
  api: {
    user: { getMyRole: { key: "role" } },
    useUtils: () => ({}),
    notifications: {
      identity: {
        key: "identity",
        useQuery: (input: unknown, options: object) =>
          useQuery({

            queryFn: () => state.identity(),

          return {

            refetch: async () => undefined,

      },

      read: { useMutation: () => ({}) },
    },
  },

let client: QueryClient;
let stop: (() => void) | undefined;
const roleKey = ["role", null];
const role = (accountKey: string | null) => ({
  notificationIdentity: {
    accountKey,
    userId: "user",
    organizationId: "org",
    actorId: "user",
    simulationId: null,
  },
});
beforeEach(() => {
  state.authority = {};
  state.inputs = [];
  state.identity.mockReset().mockResolvedValue({ accountKey: "user:org" });
  client = new QueryClient({
    defaultOptions: { queries: { retry: false, gcTime: Infinity } },
  });
  Object.defineProperty(document, "visibilityState", {
    configurable: true,
    value: "visible",
  });
  setAppPageActive(true);
});
afterEach(() => {
  cleanup();
  stop?.();
  stop = undefined;
  client.clear();
});
it("wires real standalone HTTP receipts and shared-null denial without reusing stale identity", async () => {
  render(
    <QueryClientProvider client={client}>
      <NotificationCenter sessionKey="user" />
    </QueryClientProvider>,
  );
  await screen.findByRole("button", { name: "Notifications, 2 unread" });
  client.setQueryData(roleKey, role("user:org"));
  const observer = new QueryObserver(client, {
    queryKey: roleKey,
    enabled: true,
    staleTime: Infinity,
    queryFn: async () => role(null),
  });
  act(() => {
    stop = observer.subscribe(() => undefined);
  });
  await act(async () => {
    await client.fetchQuery({
      queryKey: roleKey,
      staleTime: 0,
      queryFn: async () => role(null),
    });
  });
  expect(screen.getByRole("button", { name: "Notifications" })).toBeVisible();
  const calls = state.identity.mock.calls.length;
  expect(
    client
      .getQueryCache()
      .find({ queryKey: ["identity", { sessionKey: "user" }] })
      ?.isActive(),
  ).toBe(false);
  state.identity.mockResolvedValue({ accountKey: "new:org" });
  act(() => {
    observer.setOptions({ queryKey: roleKey, enabled: false });
  });
  await waitFor(() =>
    expect(state.identity.mock.calls.length).toBeGreaterThan(calls),
  );
  await waitFor(() => expect(state.inputs.at(-1)).toBe("new:org"));
});

it("diagnoses expiry gaps across two normal delayed 30-second role cycles", async () => {
  vi.useFakeTimers();
  const observations: { cycle: number; roleCalls: number; identityCalls: number; dialogOpen: boolean; accountKey: string | undefined }[] = [];
  const pendingRole: ((value: ReturnType<typeof role>) => void)[] = [];
  const roleRead = vi.fn(() => new Promise<ReturnType<typeof role>>(resolve => pendingRole.push(resolve)));
  render(<QueryClientProvider client={client}><NotificationCenter sessionKey="user" /></QueryClientProvider>);
  await act(async () => { await vi.advanceTimersByTimeAsync(0); });
  const observer = new QueryObserver(client, { queryKey: roleKey, enabled: true, retry: false, refetchInterval: 30_000, queryFn: roleRead });
  act(() => { stop = observer.subscribe(() => undefined); });
  await act(async () => { await vi.advanceTimersByTimeAsync(250); pendingRole.shift()!(role("user:org")); await Promise.resolve(); });
  for (let cycle = 1; cycle <= 2; cycle++) {
    // Use a delayed identity response too: observe whether normal role expiry
    // changes the keyed inbox before either fresh request can complete.
    let finishIdentity!: (value: { accountKey: string | null }) => void;
    state.identity.mockImplementationOnce(() => new Promise(resolve => { finishIdentity = resolve; }));
    const button = screen.getByRole("button", { name: "Notifications, 2 unread" });
    await act(async () => { button.click(); });
    expect(screen.getByRole("dialog", { name: "Notifications" })).toBeVisible();
    await act(async () => { await vi.advanceTimersByTimeAsync(30_000); });
    observations.push({ cycle, roleCalls: roleRead.mock.calls.length, identityCalls: state.identity.mock.calls.length,
      dialogOpen: !!screen.queryByRole("dialog", { name: "Notifications" }), accountKey: state.inputs.at(-1) });
    await act(async () => {
      await vi.advanceTimersByTimeAsync(250);
      pendingRole.splice(0).forEach(resolve => resolve(role("user:org")));
      finishIdentity({ accountKey: "user:org" });
      await Promise.resolve();
    });
  }
  console.log("notification-role-cycle-diagnostic", JSON.stringify(observations));
  expect(observations).toEqual([
    { cycle: 1, roleCalls: 2, identityCalls: 2, dialogOpen: false, accountKey: "" },
    { cycle: 2, roleCalls: 3, identityCalls: 3, dialogOpen: true, accountKey: "user:org" },
  ]);
  vi.useRealTimers();
});
